  }

  /* ---------------------------------------------------------------- minimal GLB loader (Blender export, flat colours) */
  async function loadGLB(url) {
    const r = await fetch(url); if (!r.ok) throw new Error(`${url} ${r.status}`);
    const buf = await r.arrayBuffer(); const dv = new DataView(buf); if (dv.getUint32(0, true) !== 0x46546c67) throw new Error('not a GLB');
    let off = 12, json = null, bin = null;
    while (off < buf.byteLength) { const len = dv.getUint32(off, true), type = dv.getUint32(off + 4, true); const chunk = buf.slice(off + 8, off + 8 + len); if (type === 0x4e4f534a) json = JSON.parse(new TextDecoder().decode(chunk)); else if (type === 0x004e4942) bin = chunk; off += 8 + len; }
    const acc = (i) => { const a = json.accessors[i], bv = json.bufferViews[a.bufferView]; const C = { 5120: Int8Array, 5121: Uint8Array, 5122: Int16Array, 5123: Uint16Array, 5125: Uint32Array, 5126: Float32Array }[a.componentType]; const n = { SCALAR: 1, VEC2: 2, VEC3: 3, VEC4: 4 }[a.type]; const s = (bv.byteOffset || 0) + (a.byteOffset || 0); return { array: new C(bin.slice(s, s + a.count * n * C.BYTES_PER_ELEMENT)), n }; };
    const toHex = (f) => '#' + f.slice(0, 3).map((v) => Math.round(Math.pow(Math.max(0, v), 1 / 2.2) * 255).toString(16).padStart(2, '0')).join('');
    const meshes = json.meshes.map((m) => m.primitives.map((p) => { const geo = new T.BufferGeometry(); geo.setAttribute('position', new T.BufferAttribute(acc(p.attributes.POSITION).array, 3)); if (p.attributes.NORMAL !== undefined) geo.setAttribute('normal', new T.BufferAttribute(acc(p.attributes.NORMAL).array, 3)); else geo.computeVertexNormals(); if (p.indices !== undefined) geo.setIndex(new T.BufferAttribute(acc(p.indices).array, 1)); const mat = json.materials && json.materials[p.material]; const hex = toHex((mat && mat.pbrMetallicRoughness && mat.pbrMetallicRoughness.baseColorFactor) || [0.8, 0.8, 0.8, 1]); return new T.Mesh(geo, solid('tw-model-' + hex.slice(1), hex)); }));
    const build = (ni) => { const n = json.nodes[ni]; const g = new T.Group(); g.name = n.name || ''; if (n.translation) g.position.fromArray(n.translation); if (n.rotation) g.quaternion.fromArray(n.rotation); if (n.scale) g.scale.fromArray(n.scale); if (n.mesh !== undefined) meshes[n.mesh].forEach((m) => g.add(m.clone())); if (/prop/i.test(g.name)) g.userData.prop = true; (n.children || []).forEach((ci) => g.add(build(ci))); return g; };
    const root = new T.Group(); (json.scenes[json.scene || 0].nodes || []).forEach((ni) => root.add(build(ni))); return root;
  }

  function guessCategory(ev) {

    if (/investor|vc|fund|lp |seed|series|capital|fintech|pitch/.test(t)) return 'investors';
    if (/climate|bio|health|energy|hardware/.test(t)) return 'climate';
    if (/crypto|web3|onchain|defi|token/.test(t)) return 'web3';

    if (/agent|\bai\b|model|llm|world/.test(t)) return 'ai';

  }

  let lastT = performance.now();
  function frame() {
    const now = performance.now(); const dt = Math.min(0.05, (now - lastT) / 1000); lastT = now; const t = now / 1000; const cam = c.camera;
    const animateShip = (s) => {
      const u = s.userData;
      if (u.move) { u.move.t = Math.min(1, u.move.t + dt / u.move.dur); const k = ease(u.move.t); s.position.lerpVectors(u.move.from, u.move.to, k); s.position.y += Math.sin(k * Math.PI) * 70; if (u.move.t >= 1) u.move = null; }
      else s.position.y = u.baseY + Math.sin(t * 0.45 + u.phase) * 1.8;
      const dx = cam.position.x - s.position.x, dz = cam.position.z - s.position.z;
      let err = wrapAngle((u.faceZ ? Math.atan2(dx, dz) : Math.atan2(-dz, dx)) - s.rotation.y); if (Math.abs(err) > Math.PI / 2) err = wrapAngle(err + Math.PI);
      s.rotation.y += clamp(err, -0.32 * dt, 0.32 * dt); s.rotation.z = Math.sin(t * 0.5 + u.phase) * 0.02; s.rotation.x = Math.sin(t * 0.33 + u.phase) * 0.02;
      if (u.prop) u.prop.rotation.z += dt * 9;
      if (u.tether) { const top = s.position.y - (u.tetherTop || 7), len = Math.max(1, top - u.anchorY); u.tether.scale.set(1, len, 1); u.tether.position.set(s.position.x, u.anchorY + len / 2, s.position.z); }
    };
    for (const ev of state.events) { if (ev.ship) animateShip(ev.ship); if (ev.venueGroup && ev.venueGroup.userData.flag) ev.venueGroup.userData.flag.rotation.y = Math.sin(t * 2.2 + ev.ship.userData.phase) * 0.18; }
    if (state.flagship) animateShip(state.flagship);
    if (state.mast) state.mast.userData.pennants.children.forEach((f) => { f.rotation.y = Math.sin(t * 2 + f.userData.phase) * 0.22; });
    if (state.selected && state.selected.venueGroup && state.selected.venueGroup.userData.ring) { const r = state.selected.venueGroup.userData.ring; const k = 1 + Math.sin(t * 3) * 0.08; r.scale.set(k, 1, k); }
  }

  function normalizeAddress(t) {
    let e = t.normalize('NFKC').toUpperCase().replace(/[\uFF0C,.#]/g, ' ').replace(/\s+/g, ' ').trim();
    e = e.replace(/\s+(?:APT|APARTMENT|UNIT|STE|SUITE)\s+[A-Z0-9-]+.*$/, '').replace(/\s+(?:SAN FRANCISCO|SF)(?:\s+CA(?:LIFORNIA)?)?(?:\s+94\d{3})?.*$/, '').replace(/\s+CA(?:LIFORNIA)?(?:\s+94\d{3})?.*$/, '').trim();
    for (const [a, b] of [[/\bSTREET\b/g, 'ST'], [/\bAVENUE\b/g, 'AVE'], [/\bBOULEVARD\b/g, 'BLVD'], [/\bROAD\b/g, 'RD'], [/\bDRIVE\b/g, 'DR'], [/\bCOURT\b/g, 'CT'], [/\bPLACE\b/g, 'PL'], [/\bTERRACE\b/g, 'TER'], [/\bHIGHWAY\b/g, 'HWY'], [/\bLANE\b/g, 'LN']]) e = e.replace(a, b);

  async function refreshEvents() {                             // every 15 min: re-read events.json (scrape-techweek.mjs runs daily) and merge what changed
    if (!/^(https?):/.test(location.protocol)) return;
    const data = await backend.events(); if (/^sample programme/.test(data.source)) return;
    let changed = false, structural = false;
    for (const raw of data.list) {
      const ev = state.events.find((e) => e.id === raw.id) || state.past.find((e) => e.id === raw.id);
      if (!ev) { state.events.push(normalizeEvent(raw, state.events.length)); changed = structural = true; continue; }
      const fresh = normalizeEvent(raw, 0); const keep = ev.claimedBy ? ['lat', 'lng', 'address', 'venue', 'neighborhood', 'claimed'] : [];
      for (const k of ['title', 'host', 'cohosts', 'speakers', 'start', 'end', 'startDate', 'endDate', 'dayKey', 'status', 'capacity', 'rsvp', 'url', 'image', 'description', 'category', 'venue', 'address', 'neighborhood', 'lat', 'lng', 'claimed']) {
        if (keep.includes(k)) continue; if (JSON.stringify(fresh[k]) !== JSON.stringify(ev[k])) { if (['lat', 'lng', 'claimed', 'category'].includes(k)) structural = true; ev[k] = fresh[k]; changed = true; }
      }
    }
    state.source = data.source; if (data.citizens && data.citizens.length) state.citizens = data.citizens;
    if (changed) { computeMeta(); if (state.events.some((e) => e.wantsVehicle !== !!e.hasVehicle)) structural = true; }

    if (changed) { renderList(); renderBoard(); if (state.selected) renderCard(state.selected); log('events refreshed'); }
    lifecycleTick();
  }
